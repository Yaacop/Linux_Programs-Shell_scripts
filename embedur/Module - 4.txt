#!/bin/bash

INPUT="$1"
OUTPUT="output.txt"

# Check input
if [ -z "$INPUT" ] || [ ! -f "$INPUT" ]; then
    echo "Usage: $0 input.txt"
    exit 1
fi

> "$OUTPUT"   # clear output file

while IFS= read -r line
do
    case "$line" in
        *frame.time*)
            val=$(echo "$line" | cut -d':' -f2- | xargs)
            echo "\"frame.time\": \"$val\"," >> "$OUTPUT"
            ;;

        *wlan.fc.type*)
            val=$(echo "$line" | cut -d':' -f2- | xargs)
            echo "\"wlan.fc.type\": \"$val\"," >> "$OUTPUT"
            ;;

        *wlan.fc.subtype*)
            val=$(echo "$line" | cut -d':' -f2- | xargs)
            echo "\"wlan.fc.subtype\": \"$val\"" >> "$OUTPUT"
            ;;
    esac

done < "$INPUT"

echo "Extraction complete → output.txt"
